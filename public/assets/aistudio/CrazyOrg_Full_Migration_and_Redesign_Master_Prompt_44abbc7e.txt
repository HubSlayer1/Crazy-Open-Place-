Use this Node.js signing pattern:

```typescript
import {
  createHash,
  createHmac,
  randomBytes
} from "node:crypto";

export async function callD1Worker(
  path: string,
  payload: unknown
) {
  const workerUrl = process.env.D1_WORKER_URL;
  const secret = process.env.D1_BRIDGE_SECRET;

  if (!workerUrl || !secret) {
    throw new Error("D1_BRIDGE_CONFIGURATION_MISSING");
  }

  const body = JSON.stringify(payload ?? {});
  const timestamp = Date.now().toString();
  const nonce = randomBytes(16).toString("hex");

  const bodyHash = createHash("sha256")
    .update(body)
    .digest("hex");

  const canonical = [
    timestamp,
    nonce,
    "POST",
    path,
    bodyHash
  ].join("\n");

  const signature = createHmac("sha256", secret)
    .update(canonical)
    .digest("hex");

  const response = await fetch(new URL(path, workerUrl), {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-Bridge-Timestamp": timestamp,
      "X-Bridge-Nonce": nonce,
      "X-Bridge-Signature": signature
    },
    body
  });

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`D1_BRIDGE_ERROR_${response.status}: ${errorText.slice(0, 500)}`);
  }

  return response.json();
}
```
